uint8_t GameCurrentState = 0; 

while(app open){

    if(GameCurrentState == 0){
        //intro screen
        if(user action correct){ 
            GameCurrentState++; 
        }
    }

    if(GameCurrentState == 1){
        //phys game
        if(game_flag stopped){
            if(if mario dead){ 
                GameCurrentState=2; 
            }else{
                GameCurrentState=3; 
            }
        }
    }

    if(GameCurrentState == 2){
        //game over screen
        if(user action restart){ 
            restart level
            GameCurrentState=1; 
        }

        if(user close) { close program }
    }

    if(GameCurrentState == 3){
        //next level screen
        if(user action next){ 
            switch level
            GameCurrentState=1; 
        }
    }

}
